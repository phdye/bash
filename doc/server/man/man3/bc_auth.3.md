# bc\_auth(3) -- Authenticate with bash-server

# SYNOPSIS

    #include "bashclient.h"

    int bc_auth(bc_client_t *c, const char *token);

# DESCRIPTION

Authenticates the client connection `c` with the `bash-server(1)` instance
using the provided `token`.  This function must be called exactly once
after establishing a connection with `bc_connect(3)` (or any other
connection function) and before issuing any commands.

The `token` is a 64-character hexadecimal string representing a 256-bit
random value generated by the server at startup from `/dev/urandom`.  The
server performs constant-time comparison to prevent timing attacks.

On the v2 protocol, authentication is sent as a JSON message on
`CHAN_CONTROL` (channel 0) with type `"auth"`.  The server responds with
either `"auth_ok"` or `"auth_fail"`.

If authentication fails, the connection remains open but in an
unauthenticated state.  The caller may retry with a different token or
close the connection with `bc_close(3)`.

# PARAMETERS

| Parameter | Type              | Description                                      |
|-----------|-------------------|--------------------------------------------------|
| `c`       | `bc_client_t *`   | Client handle from a `bc_connect*` function.     |
| `token`   | `const char *`    | 64-character hex string (256-bit authentication token). |

# RETURN VALUE

Returns **`BC_OK`** (0) on successful authentication.

Returns an error code on failure:

| Code              | Value | Meaning                                     |
|-------------------|-------|---------------------------------------------|
| `BC_ERR_AUTH`     | -1    | Authentication rejected by the server.      |
| `BC_ERR_PROTOCOL` | -2    | Unexpected response from the server.        |
| `BC_ERR_TIMEOUT`  | -3    | Server did not respond within the timeout.  |
| `BC_ERR_TRANSPORT`| -4    | Connection lost during authentication.      |
| `BC_ERR_PARAM`    | -7    | `c` is NULL or `token` is NULL/invalid.     |

# ERRORS

The function may fail when:

- `c` is NULL or not a valid client handle.
- `token` is NULL, empty, or not exactly 64 hexadecimal characters.
- The server rejects the token (wrong token).
- The connection is lost or reset during the authentication exchange.
- The server response is malformed or unrecognized.
- The operation times out waiting for the server's reply.

# EXAMPLES

# Basic authentication

```c
#include "bashclient.h"
#include <stdio.h>
#include <stdlib.h>

int main(void)
{
    bc_client_t *client = bc_connect(NULL);
    if (!client) {
        perror("bc_connect");
        return 1;
    }

    const char *token = getenv("BASH_SERVER_TOKEN");
    if (!token) {
        fprintf(stderr, "BASH_SERVER_TOKEN not set\n");
        bc_close(client);
        return 1;
    }

    int rc = bc_auth(client, token);
    if (rc != BC_OK) {
        fprintf(stderr, "authentication failed (rc=%d): %s\n",
                rc, bc_error(client));
        bc_close(client);
        return 1;
    }

    printf("authenticated successfully\n");

    /* Now safe to issue commands */
    bc_eval_result_t result;
    if (bc_eval(client, "whoami", &result) == BC_OK) {
        printf("user: %s", result.stdout_data);
        bc_eval_result_free(&result);
    }

    bc_close(client);
    return 0;
}
```

# Reading token from a file

```c
#include "bashclient.h"
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

static char *read_token_file(const char *path)
{
    FILE *fp = fopen(path, "r");
    if (!fp)
        return NULL;

    char buf[128];
    if (!fgets(buf, sizeof(buf), fp)) {
        fclose(fp);
        return NULL;
    }
    fclose(fp);

    /* Strip trailing newline */
    size_t len = strlen(buf);
    while (len > 0 && (buf[len-1] == '\n' || buf[len-1] == '\r'))
        buf[--len] = '\0';

    return strdup(buf);
}

int main(void)
{
    bc_client_t *client = bc_connect(NULL);
    if (!client) {
        perror("bc_connect");
        return 1;
    }

    char *token = read_token_file("/tmp/bash-server-1000/token");
    if (!token) {
        perror("read token");
        bc_close(client);
        return 1;
    }

    int rc = bc_auth(client, token);
    free(token);

    if (rc != BC_OK) {
        fprintf(stderr, "auth: %s\n", bc_error(client));
        bc_close(client);
        return 1;
    }

    printf("authenticated\n");
    bc_close(client);
    return 0;
}
```

# Error handling for each failure mode

```c
#include "bashclient.h"
#include <stdio.h>
#include <stdlib.h>

void authenticate(bc_client_t *client, const char *token)
{
    int rc = bc_auth(client, token);

    switch (rc) {
    case BC_OK:
        printf("auth succeeded\n");
        break;
    case BC_ERR_AUTH:
        fprintf(stderr, "wrong token\n");
        break;
    case BC_ERR_PROTOCOL:
        fprintf(stderr, "protocol error: %s\n", bc_error(client));
        break;
    case BC_ERR_TIMEOUT:
        fprintf(stderr, "server did not respond\n");
        break;
    case BC_ERR_TRANSPORT:
        fprintf(stderr, "connection lost\n");
        break;
    case BC_ERR_PARAM:
        fprintf(stderr, "invalid parameters\n");
        break;
    default:
        fprintf(stderr, "unknown error %d: %s\n", rc, bc_error(client));
        break;
    }
}
```

# NOTES

- Authentication must be performed exactly once per connection.  Calling
  `bc_auth` on an already-authenticated connection returns `BC_ERR_PROTOCOL`.

- The token is compared using a constant-time algorithm on the server side
  to prevent timing-based side-channel attacks.

- The 64-character hex token represents 256 bits of entropy from
  `/dev/urandom`.  Treat it as a secret -- do not log it, embed it in
  source code, or transmit it over insecure channels.

- If the server was started with `--token <value>`, use that exact token.
  Otherwise, read the auto-generated token from the server's token file
  (typically `<socket_dir>/token`).

- After a failed authentication attempt, the connection remains open.
  The caller may retry with a different token.  However, the server may
  impose rate limits or close the connection after repeated failures.

- The function blocks until the server responds or the timeout expires.
  The default timeout is 10 seconds.
